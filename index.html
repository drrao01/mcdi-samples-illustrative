import os
from typing import Any, Dict, Optional
import requests


def fetch_usd_to_eur_rate(timeout: float = 10.0) -> float:
  """Fetches the current USD->EUR exchange rate from the Acme FX API.

  Args:
      timeout (float): Request timeout in seconds. Defaults to 10.0.

  Returns:
      float: The exchange rate for USD to EUR.

  Raises:
      requests.RequestException: If the HTTP request fails.
      KeyError: If the expected rate key is missing from the response payload.
  """
  url: str = "https://api.acme-fx.internal/v1/rates"
  headers: Dict[str, str] = {}

  api_key: Optional[str] = os.environ.get("ACME_FX_API_KEY")
  if api_key:
    headers["Authorization"] = f"Bearer {api_key}"

  response: requests.Response = requests.get(
      url,
      params={"base": "USD", "target": "EUR"},
      headers=headers,
      timeout=timeout,
  )
  response.raise_for_status()

  data: Dict[str, Any] = response.json()

  if "rate" in data:
    return float(data["rate"])
  elif "rates" in data and isinstance(data["rates"], dict) and "EUR" in data["rates"]:
    return float(data["rates"]["EUR"])

  raise KeyError("Exchange rate for USD->EUR not found in API response.")

import os
import time
from typing import Any, Dict, Optional
import httpx


def fetch_usd_to_eur_rate(
    max_retries: int = 3,
    initial_backoff: float = 1.0,
    timeout: float = 10.0,
) -> float:
  """Fetches the current USD->EUR exchange rate from the Acme FX API with retry logic and exponential backoff.

  Args:
      max_retries (int): Maximum number of retry attempts. Defaults to 3.
      initial_backoff (float): Base wait time in seconds for exponential
        backoff. Defaults to 1.0.
      timeout (float): Request timeout in seconds for HTTP requests. Defaults
        to 10.0.

  Returns:
      float: The exchange rate for USD to EUR.

  Raises:
      httpx.HTTPError: If the HTTP request fails after all retry attempts.
      KeyError: If the expected rate key is missing from the response payload.
  """
  url: str = "https://api.acme-fx.internal/v1/rates"
  headers: Dict[str, str] = {}

  api_key: Optional[str] = os.environ.get("ACME_FX_API_KEY")
  if api_key:
    headers["Authorization"] = f"Bearer {api_key}"

  last_exception: Optional[Exception] = None

  for attempt in range(max_retries):
    try:
      with httpx.Client(timeout=timeout) as client:
        response: httpx.Response = client.get(
            url,
            params={"base": "USD", "target": "EUR"},
            headers=headers,
        )
        response.raise_for_status()

        data: Dict[str, Any] = response.json()

        if "rate" in data:
          return float(data["rate"])
        elif (
            "rates" in data
            and isinstance(data["rates"], dict)
            and "EUR" in data["rates"]
        ):
          return float(data["rates"]["EUR"])

        raise KeyError("Exchange rate for USD->EUR not found in API response.")
    except httpx.HTTPError as exc:
      last_exception = exc
      if attempt < max_retries - 1:
        sleep_time: float = initial_backoff * (2**attempt)
        time.sleep(sleep_time)

  if last_exception is not None:
    raise last_exception

  raise RuntimeError("Failed to fetch exchange rate after max retries.")
