import logging
import os
from typing import Any, Dict, Optional
import requests

logger = logging.getLogger(__name__)


def redact_text(text: str, secret: Optional[str]) -> str:
    """Redact secret credentials from a text string if present."""
    if secret and secret in text:
        return text.replace(secret, "[REDACTED]")
    return text


def extract_eur_rate(payload: Any) -> float:
    """Extract the USD->EUR rate as a float from various JSON payload structures."""
    if isinstance(payload, (int, float)):
        return float(payload)
    if isinstance(payload, dict):
        if "rates" in payload and isinstance(payload["rates"], dict):
            rates = payload["rates"]
            if "EUR" in rates:
                return float(rates["EUR"])
            if "USD_EUR" in rates:
                return float(rates["USD_EUR"])
        if "data" in payload:
            try:
                return extract_eur_rate(payload["data"])
            except ValueError:
                pass
        if "EUR" in payload:
            return float(payload["EUR"])
        if "USD_EUR" in payload:
            return float(payload["USD_EUR"])
        if "rate" in payload:
            return float(payload["rate"])
    raise ValueError("Unable to parse USD->EUR rate from response payload")


def fetch_usd_to_eur_rate(
    endpoint: str = "https://api.acme-fx.internal/v1/rates",
    timeout: float = 10.0,
) -> float:
    """Fetch the USD to EUR exchange rate from the Acme FX API endpoint.

    Reads the API key from the ACME_FX_API_KEY environment variable.

    Args:
        endpoint: URL of the Acme FX rates endpoint.
        timeout: Request timeout in seconds.

    Returns:
        The USD->EUR exchange rate as a float.

    Raises:
        ValueError: If ACME_FX_API_KEY is missing or the response cannot be parsed.
        RuntimeError: If the HTTP request fails.
    """
    api_key: Optional[str] = os.environ.get("ACME_FX_API_KEY")
    if not api_key:
        raise ValueError("ACME_FX_API_KEY environment variable is not set")

    headers: Dict[str, str] = {
        "X-Api-Key": api_key,
        "Accept": "application/json",
    }
    params: Dict[str, str] = {
        "base": "USD",
        "symbols": "EUR",
    }

    try:
        response: requests.Response = requests.get(
            endpoint,
            headers=headers,
            params=params,
            timeout=timeout,
        )
        response.raise_for_status()
        payload: Any = response.json()
    except requests.RequestException as exc:
        clean_msg: str = redact_text(str(exc), api_key)
        logger.error("Failed to fetch USD->EUR rate: {}".format(clean_msg))
        raise RuntimeError("Failed to fetch USD->EUR rate: {}".format(clean_msg)) from None
    except Exception as exc:
        clean_msg: str = redact_text(str(exc), api_key)
        logger.error("Unexpected error fetching USD->EUR rate: {}".format(clean_msg))
        raise RuntimeError("Unexpected error fetching exchange rate: {}".format(clean_msg)) from None

    try:
        rate: float = extract_eur_rate(payload)
    except ValueError as exc:
        clean_msg: str = redact_text(str(exc), api_key)
        raise ValueError("Invalid rate payload: {}".format(clean_msg)) from None

    return rate


def get_usd_to_eur_rate(
    endpoint: str = "https://api.acme-fx.internal/v1/rates",
    timeout: float = 10.0,
) -> float:
    """Alias function for fetch_usd_to_eur_rate."""
    return fetch_usd_to_eur_rate(endpoint=endpoint, timeout=timeout)

def fetch_usd_to_eur_rate(
    endpoint: str = "https://api.acme-fx.internal/v1/rates",
    timeout: float = 10.0,
    max_retries: int = 3,
    backoff_factor: float = 1.0,
) -> float:
    """Fetch the USD to EUR exchange rate from the Acme FX API.

    Reads the API key from the ACME_FX_API_KEY environment variable.
    Includes 3-attempt retry logic with exponential backoff.

    Args:
        endpoint: URL of the Acme FX rates endpoint.
        timeout: Request timeout in seconds.
        max_retries: Maximum number of request attempts.
        backoff_factor: Multiplier for exponential backoff sleep duration.

    Returns:
        The USD->EUR exchange rate as a float.

    Raises:
        ValueError: If ACME_FX_API_KEY is not set or response payload is invalid.
        RuntimeError: If the HTTP request fails after all retry attempts.
    """
    api_key: Optional[str] = os.environ.get("ACME_FX_API_KEY")
    if not api_key:
        raise ValueError("ACME_FX_API_KEY environment variable is not set")

    headers: Dict[str, str] = {
        "X-Api-Key": api_key,
        "Accept": "application/json",
    }
    params: Dict[str, str] = {
        "base": "USD",
        "symbols": "EUR",
    }

    last_exception: Optional[Exception] = None

    for attempt in range(max_retries):
        try:
            response: requests.Response = requests.get(
                endpoint,
                headers=headers,
                params=params,
                timeout=timeout,
            )
            response.raise_for_status()
            payload: Any = response.json()
            rate: float = extract_eur_rate(payload)
            return rate
        except requests.RequestException as exc:
            clean_msg: str = redact_text(str(exc), api_key)
            logger.warning(
                "Attempt {}/{} failed to fetch exchange rate: {}".format(
                    attempt + 1, max_retries, clean_msg
                )
            )
            last_exception = exc
            if attempt < max_retries - 1:
                sleep_time: float = backoff_factor * (2 ** attempt)
                time.sleep(sleep_time)
        except ValueError as exc:
            clean_msg: str = redact_text(str(exc), api_key)
            logger.error("Failed to parse USD->EUR rate payload: {}".format(clean_msg))
            raise ValueError("Invalid rate payload: {}".format(clean_msg)) from None
        except Exception as exc:
            clean_msg: str = redact_text(str(exc), api_key)
            logger.error("Unexpected error fetching USD->EUR rate: {}".format(clean_msg))
            raise RuntimeError("Unexpected error fetching exchange rate: {}".format(clean_msg)) from None

    clean_last_msg: str = redact_text(str(last_exception), api_key) if last_exception else "Max retries exceeded"
    logger.error("All {} attempts to fetch USD->EUR rate failed: {}".format(max_retries, clean_last_msg))
    raise RuntimeError("Failed to fetch USD->EUR rate after {} attempts: {}".format(max_retries, clean_last_msg)) from None

def fetch_usd_to_eur_rate(
    endpoint: str = "https://api.acme-fx.internal/v1/rates",
    timeout: float = 10.0,
    max_retries: int = 3,
    backoff_factor: float = 1.0,

    """Fetch the USD to EUR exchange rate from the Acme FX API.

    Reads the API key from the ACME_FX_API_KEY environment variable.
    Includes 3-attempt retry logic with exponential backoff.

    Args:
        endpoint: URL of the Acme FX rates endpoint.
        timeout: Request timeout in seconds.
        max_retries: Maximum number of request attempts.
        backoff_factor: Multiplier for exponential backoff sleep duration.

    Returns:
        The USD->EUR exchange rate as a float.

    Raises:
        ValueError: If ACME_FX_API_KEY is not set or response payload is invalid.
        RuntimeError: If the HTTP request fails after all retry attempts.
    """

                endpoint,
                headers=headers,
                params=params,
                timeout=timeout,
            )
            response.raise_for_status()
            payload: Any = response.json()

    timeout: float = 10.0,
    max_retries: int = 3,
    backoff_factor: float = 1.0,

        endpoint=endpoint,
        timeout=timeout,
        max_retries=max_retries,

def convert_usd_to_eur(
    amount_usd: float,
    rate: Optional[float] = None,
    endpoint: str = "https://api.acme-fx.internal/v1/rates",
) -> float:
    """Convert USD amount to EUR using the FX rate and round to 2 decimal places (cents).

    Args:
        amount_usd: Amount in USD to convert.
        rate: Optional pre-fetched USD->EUR exchange rate. If None, fetches rate from API.
        endpoint: Endpoint URL if rate needs to be fetched.

    Returns:
        Converted amount in EUR rounded to 2 decimal places (cents) as a float.
    """
    if rate is None:
        rate = fetch_usd_to_eur_rate(endpoint=endpoint)
    amt_dec: Decimal = Decimal(str(amount_usd))
    rate_dec: Decimal = Decimal(str(rate))
    eur_dec: Decimal = (amt_dec * rate_dec).quantize(Decimal("0.01"), rounding=ROUND_HALF_UP)
    return float(eur_dec)
